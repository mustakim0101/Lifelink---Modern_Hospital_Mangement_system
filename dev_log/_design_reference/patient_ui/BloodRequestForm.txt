import React, { useState } from 'react';
import { Droplet, AlertCircle, Check } from 'lucide-react';
import { AppShell } from '../../components/layout/AppShell';
import { PageHeader } from '../../components/shared/PageHeader';
import { Card, CardContent, CardHeader, CardTitle } from '../../components/ui/card';
import { Button } from '../../components/ui/button';
import { Label } from '../../components/ui/label';
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '../../components/ui/select';
import { Textarea } from '../../components/ui/textarea';
import { Input } from '../../components/ui/input';
import { patients, departments, bloodBanks } from '../../data/mockData';
import { toast } from 'sonner';

export default function BloodRequestForm() {
  const [submitted, setSubmitted] = useState(false);
  const [formData, setFormData] = useState({
    bloodGroup: '',
    component: '',
    units: '1',
    urgency: '',
    department: '',
    bloodBank: '',
    reason: '',
  });

  // Mock: using patient David Kumar
  const patient = patients[2];

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    toast.success('Blood request submitted successfully');
    setSubmitted(true);
  };

  if (submitted) {
    return (
      <AppShell>
        <PageHeader
          title="Blood Request Submitted"
          breadcrumb={[
            { label: 'Patient Portal', path: '/patient/portal' },
            { label: 'Blood Requests', path: '/patient/blood-requests' },
            { label: 'Request Submitted' },
          ]}
        />

        <Card className="max-w-2xl mx-auto border-2 border-[#059669]">
          <CardContent className="pt-12 pb-12 text-center">
            <div className="w-16 h-16 bg-[#D1FAE5] rounded-full flex items-center justify-center mx-auto mb-4">
              <Check className="w-8 h-8 text-[#059669]" />
            </div>
            <h2 className="text-2xl font-semibold mb-2">Request Submitted Successfully</h2>
            <p className="text-muted-foreground mb-2">
              Your blood request has been submitted to the blood bank.
            </p>
            <p className="text-muted-foreground mb-6">
              You will be notified once compatible donors are identified and the request is processed.
            </p>

            <div className="bg-[#E0F2FE] rounded-lg p-4 mb-6 text-left max-w-md mx-auto">
              <h3 className="font-medium mb-3">Request Summary</h3>
              <div className="space-y-2 text-sm">
                <div className="flex justify-between">
                  <span className="text-muted-foreground">Blood Group:</span>
                  <span className="font-medium text-[#DC2626]">{formData.bloodGroup}</span>
                </div>
                <div className="flex justify-between">
                  <span className="text-muted-foreground">Component:</span>
                  <span className="font-medium">{formData.component}</span>
                </div>
                <div className="flex justify-between">
                  <span className="text-muted-foreground">Units:</span>
                  <span className="font-medium">{formData.units}</span>
                </div>
                <div className="flex justify-between">
                  <span className="text-muted-foreground">Urgency:</span>
                  <span className="font-medium">{formData.urgency}</span>
                </div>
              </div>
            </div>

            <div className="flex gap-3 justify-center">
              <Button variant="outline" onClick={() => window.location.href = '/patient/blood-requests'}>
                View All Requests
              </Button>
              <Button className="bg-[#0369A1] hover:bg-[#075985]" onClick={() => window.location.href = '/patient/portal'}>
                Return to Portal
              </Button>
            </div>
          </CardContent>
        </Card>
      </AppShell>
    );
  }

  return (
    <AppShell>
      <PageHeader
        title="Request Blood"
        subtitle="Submit a blood request for medical treatment"
        breadcrumb={[
          { label: 'Patient Portal', path: '/patient/portal' },
          { label: 'Blood Requests', path: '/patient/blood-requests' },
          { label: 'New Request' },
        ]}
      />

      {/* Patient Info */}
      <Card className="mb-6 border-2 border-[#0369A1]">
        <CardContent className="pt-6">
          <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
            <div>
              <div className="text-sm text-muted-foreground">Patient Name</div>
              <div className="font-medium">{patient.name}</div>
            </div>
            <div>
              <div className="text-sm text-muted-foreground">MRN</div>
              <div className="font-medium">{patient.mrn}</div>
            </div>
            <div>
              <div className="text-sm text-muted-foreground">Your Blood Group</div>
              <div className="font-medium text-[#DC2626]">{patient.bloodGroup}</div>
            </div>
            <div>
              <div className="text-sm text-muted-foreground">Age</div>
              <div className="font-medium">{patient.age} years</div>
            </div>
          </div>
        </CardContent>
      </Card>

      {/* Important Notice */}
      <Card className="mb-6 border-2 border-[#D97706] bg-[#FEF3C7]">
        <CardContent className="pt-6">
          <div className="flex items-start gap-3">
            <AlertCircle className="w-6 h-6 text-[#D97706] flex-shrink-0 mt-0.5" />
            <div>
              <h3 className="font-medium text-[#D97706] mb-2">Important Information</h3>
              <ul className="text-sm space-y-1 text-foreground">
                <li>• Blood requests require approval from your attending physician</li>
                <li>• Emergency requests are prioritized and processed immediately</li>
                <li>• You will be notified once compatible donors are identified</li>
                <li>• All blood components undergo strict safety testing</li>
              </ul>
            </div>
          </div>
        </CardContent>
      </Card>

      {/* Request Form */}
      <form onSubmit={handleSubmit}>
        <Card className="mb-6">
          <CardHeader>
            <CardTitle>Blood Request Details</CardTitle>
          </CardHeader>
          <CardContent className="space-y-6">
            <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
              {/* Blood Group */}
              <div className="space-y-2">
                <Label htmlFor="bloodGroup">Required Blood Group *</Label>
                <Select 
                  value={formData.bloodGroup} 
                  onValueChange={(value) => setFormData({ ...formData, bloodGroup: value })}
                  required
                >
                  <SelectTrigger>
                    <SelectValue placeholder="Select blood group" />
                  </SelectTrigger>
                  <SelectContent>
                    {['A+', 'A-', 'B+', 'B-', 'AB+', 'AB-', 'O+', 'O-'].map((bg) => (
                      <SelectItem key={bg} value={bg}>{bg}</SelectItem>
                    ))}
                  </SelectContent>
                </Select>
              </div>

              {/* Component */}
              <div className="space-y-2">
                <Label htmlFor="component">Blood Component *</Label>
                <Select 
                  value={formData.component} 
                  onValueChange={(value) => setFormData({ ...formData, component: value })}
                  required
                >
                  <SelectTrigger>
                    <SelectValue placeholder="Select component" />
                  </SelectTrigger>
                  <SelectContent>
                    <SelectItem value="Whole Blood">Whole Blood</SelectItem>
                    <SelectItem value="Plasma">Plasma</SelectItem>
                    <SelectItem value="Platelets">Platelets</SelectItem>
                    <SelectItem value="RBC">Red Blood Cells (RBC)</SelectItem>
                  </SelectContent>
                </Select>
              </div>

              {/* Units */}
              <div className="space-y-2">
                <Label htmlFor="units">Number of Units *</Label>
                <Input
                  id="units"
                  type="number"
                  min="1"
                  max="10"
                  value={formData.units}
                  onChange={(e) => setFormData({ ...formData, units: e.target.value })}
                  required
                />
              </div>

              {/* Urgency */}
              <div className="space-y-2">
                <Label htmlFor="urgency">Urgency Level *</Label>
                <Select 
                  value={formData.urgency} 
                  onValueChange={(value) => setFormData({ ...formData, urgency: value })}
                  required
                >
                  <SelectTrigger>
                    <SelectValue placeholder="Select urgency" />
                  </SelectTrigger>
                  <SelectContent>
                    <SelectItem value="Normal">Normal</SelectItem>
                    <SelectItem value="Urgent">Urgent</SelectItem>
                    <SelectItem value="Emergency">Emergency</SelectItem>
                  </SelectContent>
                </Select>
              </div>

              {/* Department */}
              <div className="space-y-2">
                <Label htmlFor="department">Department *</Label>
                <Select 
                  value={formData.department} 
                  onValueChange={(value) => setFormData({ ...formData, department: value })}
                  required
                >
                  <SelectTrigger>
                    <SelectValue placeholder="Select department" />
                  </SelectTrigger>
                  <SelectContent>
                    {departments.map((dept) => (
                      <SelectItem key={dept.id} value={dept.name}>{dept.name}</SelectItem>
                    ))}
                  </SelectContent>
                </Select>
              </div>

              {/* Blood Bank */}
              <div className="space-y-2">
                <Label htmlFor="bloodBank">Preferred Blood Bank</Label>
                <Select 
                  value={formData.bloodBank} 
                  onValueChange={(value) => setFormData({ ...formData, bloodBank: value })}
                >
                  <SelectTrigger>
                    <SelectValue placeholder="Select blood bank (optional)" />
                  </SelectTrigger>
                  <SelectContent>
                    {bloodBanks.map((bb) => (
                      <SelectItem key={bb.id} value={bb.name}>{bb.name}</SelectItem>
                    ))}
                  </SelectContent>
                </Select>
              </div>
            </div>

            {/* Reason */}
            <div className="space-y-2">
              <Label htmlFor="reason">Medical Reason / Context *</Label>
              <Textarea
                id="reason"
                placeholder="Please provide details about why blood is needed (e.g., surgery, treatment, emergency)"
                value={formData.reason}
                onChange={(e) => setFormData({ ...formData, reason: e.target.value })}
                rows={4}
                required
              />
            </div>
          </CardContent>
        </Card>

        {/* Submit */}
        <div className="flex gap-3">
          <Button 
            type="button" 
            variant="outline" 
            className="flex-1"
            onClick={() => window.location.href = '/patient/blood-requests'}
          >
            Cancel
          </Button>
          <Button type="submit" className="flex-1 bg-[#DC2626] hover:bg-[#B91C1C]">
            <Droplet className="w-4 h-4 mr-2" />
            Submit Blood Request
          </Button>
        </div>
      </form>
    </AppShell>
  );
}
